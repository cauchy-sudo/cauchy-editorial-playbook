% ------- steps: name / colour / y / height / title / purpose -------
\newcommand{\stepnode}[6]{\node[step=#2, minimum height=#4 cm] (#1) at (\xs,#3) {\textbf{#5}\\[1pt]#6};}

\stepnode{s1}{phA}{\ya}{1.15}{Verify the sample}
  {Keep only stocks listed in every official Index X evaluation, Feb.\ 2022--Dec.\ 2025.}
\stepnode{s2}{phA}{\yb}{1.15}{Collect price data}
  {Download daily adjusted closing prices $S_t$ for the verified stocks.}
\stepnode{s3}{phA}{\yc}{1.15}{Compute daily returns}
  {Convert prices to log returns, $r_t=\ln(S_t/S_{t-1})$.}

\stepnode{s4}{phB}{\yd}{1.15}{Test the return series}
  {ADF: are returns stationary? ARCH-LM: does volatility cluster over time?}
\stepnode{s5}{phB}{\ye}{1.15}{Fit GJR-GARCH(1,1)}
  {Model time-varying volatility $\sigma_t$; the term $\gamma$ lets negative shocks raise it more.}
\stepnode{s6}{phB}{\yf}{1.15}{Forecast the base volatility}
  {Annualize the one-day-ahead forecast:\\ \mbox{$\sigma_{\mathrm{ann}}=\sigma_{\mathrm{daily}}\sqrt{252}$}.}

\stepnode{s7}{phC}{\yg}{2.5}{Generate option-contract scenarios}
  {Draw each input independently and uniformly, using a Sobol sequence, so that the scenarios cover the whole input space.}
\stepnode{s8}{phC}{\yh}{1.15}{Price every scenario}
  {Black--Scholes call value, using $\sigma^{*}=m_\sigma\sigma_{\mathrm{ann}}$ and scaled by $(1+\tau)$.}

\stepnode{s9}{phD}{\yi}{1.15}{Decompose the output variance}
  {Sobol analysis: how much of the variation in $Y$ is caused by each input?}
\stepnode{s10}{phD}{\yj}{1.15}{Check numerical reliability}
  {Repeat with more samples and resample across stocks (bootstrap).}
\stepnode{s11}{phD}{\yk}{1.15}{Test robustness}
  {Re-run with other input ranges and error distributions to see if the ranking survives.}
\stepnode{s12}{phD}{\yl}{1.15}{Rank the input factors}
  {Average the indices over all stocks and order the factors by importance.}
